class GalaxyPlayer : public CharacterBody2D { private: float speed = 450.0f; float jump_force = -800.0f; public: void _physics_process(float delta) override { Vector2 velocity = get_velocity(); // Control de movimiento horizontal fluido float input_dir = Input.get_axis("ui_left", "ui_right"); if (input_dir != 0) { velocity.x = input_dir * speed; } else { velocity.x = move_toward(velocity.x, 0, speed * delta); } set_velocity(velocity); move_and_slide(); } };
